% Screenshot helper: includes figures/screenshots/<file>.png if present,
% otherwise draws a clearly marked placeholder so the document still builds.
% Usage: \screenshot{file-name}{Caption text}{fig:label}{width-fraction}
\newcommand{\screenshot}[4]{%
  \begin{figure}[H]
    \centering
    \IfFileExists{figures/screenshots/#1.png}{%
      \includegraphics[width=#4\textwidth]{figures/screenshots/#1.png}%
    }{%
      \fbox{\begin{minipage}[c][0.40\textwidth][c]{\dimexpr#4\textwidth-2\fboxsep-2\fboxrule\relax}
        \centering\color{tcgray}
        \textbf{Screenshot placeholder}\\[0.5em]
        \texttt{figures/screenshots/\detokenize{#1}.png}\\[0.5em]
        \small #2
      \end{minipage}}%
    }
    \caption{#2}
    \label{#3}
  \end{figure}%
}
